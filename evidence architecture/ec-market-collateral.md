## EC Balance Management Logic

```javascript
// EC Wallet Balance Manager

class ECWalletManager {
  
  async consumeEC(accountId, ecAmount, evidenceDetails) {
    // 1. Check current balance
    const wallet = await this.getWallet(accountId);
    
    if (wallet.balance < ecAmount) {
      // Insufficient funds
      await this.handleInsufficientFunds(accountId, ecAmount);
      return { success: false, reason: 'insufficient_funds' };
    }
    
    // 2. Create consumption transaction
    const transaction = {
      transaction_id: generateId('txn'),
      timestamp: new Date(),
      account_id: accountId,
      amount: -ecAmount,
      balance_before: wallet.balance,
      balance_after: wallet.balance - ecAmount,
      evidence_details: evidenceDetails,
      status: 'completed'
    };
    
    // 3. Update balance
    await this.updateBalance(accountId, -ecAmount);
    await this.logTransaction(transaction);
    
    // 4. Check alert thresholds
    const newBalance = wallet.balance - ecAmount;
    if (newBalance < wallet.alert_threshold) {
      await this.triggerLowBalanceAlert(accountId, newBalance);
    }
    
    // 5. Check auto-reload
    if (wallet.auto_reload_enabled && newBalance < wallet.auto_reload_threshold) {
      await this.triggerAutoReload(accountId);
    }
    
    return { 
      success: true, 
      new_balance: newBalance,
      transaction_id: transaction.transaction_id 
    };
  }
  
  async handleInsufficientFunds(accountId, requiredEC) {
    // Pause non-critical monitoring
    await this.pauseLowPriorityMonitoring(accountId);
    
    // Send urgent alert
    await this.sendUrgentAlert(accountId, {
      type: 'insufficient_ec_funds',
      required_ec: requiredEC,
      current_balance: (await this.getWallet(accountId)).balance,
      recommended_action: 'top_up_immediately'
    });
    
    // Log event
    await this.logEvent(accountId, 'ec_insufficient_funds', {
      required: requiredEC,
      timestamp: new Date()
    });
  }
  
  async triggerAutoReload(accountId) {
    const wallet = await this.getWallet(accountId);
    
    if (!wallet.auto_reload_enabled) return;
    
    // Create purchase order
    const purchase = {
      purchase_id: generateId('pur'),
      account_id: accountId,
      package: wallet.auto_reload_package,
      ec_amount: wallet.auto_reload_ec_amount,
      price: this.calculatePrice(wallet.auto_reload_package),
      payment_method: wallet.default_payment_method,
      triggered_by: 'auto_reload',
      timestamp: new Date()
    };
    
    // Process payment
    const paymentResult = await this.processPayment(purchase);
    
    if (paymentResult.success) {
      // Add EC to wallet
      await this.addEC(accountId, purchase.ec_amount, {
        source: 'auto_reload',
        purchase_id: purchase.purchase_id
      });
      
      // Send confirmation
      await this.sendAutoReloadConfirmation(accountId, purchase);
    } else {
      // Payment failed - alert user
      await this.sendPaymentFailureAlert(accountId, paymentResult);
    }
  }
  
  async calculateDailyUsage(accountId, days = 30) {
    const transactions = await this.getTransactions(accountId, {
      days: days,
      type: 'consumption'
    });
    
    const totalEC = transactions.reduce((sum, txn) => sum + Math.abs(txn.amount), 0);
    const dailyAverage = totalEC / days;
    
    return {
      total_ec_consumed: totalEC,
      daily_average: dailyAverage,
      days_analyzed: days
    };
  }
  
  async estimateRunwayDays(accountId) {
    const wallet = await this.getWallet(accountId);
    const usage = await this.calculateDailyUsage(accountId, 30);
    
    if (usage.daily_average === 0) {
      return { runway_days: Infinity, message: 'No recent usage' };
    }
    
    const runwayDays = Math.floor(wallet.balance / usage.daily_average);
    
    return {
      runway_days: runwayDays,
      current_balance: wallet.balance,
      daily_burn_rate: usage.daily_average,
      estimated_depletion_date: new Date(Date.now() + (runwayDays * 24 * 60 * 60 * 1000))
    };
  }
}
```

```javascript
// EC Package Pricing Engine

class ECPricingCalculator {
  
  constructor() {
    this.baseECPrice = 1.00; // $1 per EC at base
    this.packages = {
      'monitoring_starter': { ec: 500, discount: 0.20, price: 400 },
      'monitoring_growth': { ec: 2500, discount: 0.28, price: 1800 },
      'monitoring_scale': { ec: 10000, discount: 0.35, price: 6500 },
      'portfolio_essentials': { ec: 25000, discount: 0.40, price: 15000 },
      'portfolio_professional': { ec: 100000, discount: 0.50, price: 50000 },
      'portfolio_enterprise': { ec: 500000, discount: 0.60, price: 200000 }
    };
  }
  
  calculateCustomPrice(ecAmount) {
    // Volume discount tiers
    let discount = 0;
    
    if (ecAmount >= 500000) discount = 0.60;
    else if (ecAmount >= 100000) discount = 0.50;
    else if (ecAmount >= 25000) discount = 0.40;
    else if (ecAmount >= 10000) discount = 0.35;
    else if (ecAmount >= 2500) discount = 0.28;
    else if (ecAmount >= 500) discount = 0.20;
    else if (ecAmount >= 100) discount = 0.10;
    
    const basePrice = ecAmount * this.baseECPrice;
    const discountedPrice = basePrice * (1 - discount);
    
    return {
      ec_amount: ecAmount,
      base_price: basePrice,
      discount_percent: discount * 100,
      discount_amount: basePrice - discountedPrice,
      final_price: discountedPrice,
      price_per_ec: discountedPrice / ecAmount
    };
  }
  
  recommendPackage(estimatedMonthlyUsage, months = 12) {
    const totalEC = estimatedMonthlyUsage * months;
    
    // Find best matching package
    let recommendedPackage = null;
    let bestValue = Infinity;
    
    for (const [key, pkg] of Object.entries(this.packages)) {
      if (pkg.ec >= totalEC * 0.8) { // At least 80% coverage
        const valueRatio = pkg.price / pkg.ec;
        if (valueRatio < bestValue) {
          bestValue = valueRatio;
          recommendedPackage = { name: key, ...pkg };
        }
      }
    }
    
    return {
      estimated_annual_ec: totalEC,
      recommended_package: recommendedPackage,
      coverage_months: recommendedPackage ? (recommendedPackage.ec / estimatedMonthlyUsage) : 0,
      annual_cost: recommendedPackage ? recommendedPackage.price : this.calculateCustomPrice(totalEC).final_price
    };
  }
  
  compareROI(portfolioSize, traditionalSiteVisitCost = 35000, visitsPerYear = 2) {
    // Traditional model
    const traditionalAnnualCost = portfolioSize * traditionalSiteVisitCost * visitsPerYear;
    
    // FutureForm EC model
    let ecPackage;
    if (portfolioSize <= 5) ecPackage = this.packages.monitoring_growth;
    else if (portfolioSize <= 20) ecPackage = this.packages.portfolio_essentials;
    else if (portfolioSize <= 50) ecPackage = this.packages.portfolio_professional;
    else ecPackage = this.packages.portfolio_enterprise;
    
    // Assume ~$35K for Enterprise RC program
    const futureformCost = 35000 + ecPackage.price;
    
    const savings = traditionalAnnualCost - futureformCost;
    const savingsPercent = (savings / traditionalAnnualCost) * 100;
    
    return {
      portfolio_size: portfolioSize,
      traditional_model: {
        annual_cost: traditionalAnnualCost,
        frequency: `${visitsPerYear}x per year per company`,
        total_visits: portfolioSize * visitsPerYear
      },
      futureform_model: {
        annual_cost: futureformCost,
        frequency: 'Continuous daily monitoring',
        package: ecPackage
      },
      savings: {
        annual_amount: savings,
        percentage: savingsPercent.toFixed(1),
        roi_multiple: (traditionalAnnualCost / futureformCost).toFixed(1)
      }
    };
  }
}
```
